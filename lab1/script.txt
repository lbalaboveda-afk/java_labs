
CREATE SEQUENCE seq_artist_id START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE seq_album_id START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE seq_composition_id START WITH 1 INCREMENT BY 1;

CREATE TABLE Artist (
    artist_id     INTEGER      PRIMARY KEY DEFAULT nextval('seq_artist_id'),
    artist_name   VARCHAR(100) NOT NULL UNIQUE,
    CONSTRAINT chk_artist_name CHECK (LENGTH(TRIM(artist_name)) > 0)
);
ALTER SEQUENCE seq_artist_id OWNED BY Artist.artist_id;

CREATE TABLE Album (
    album_id      INTEGER      PRIMARY KEY DEFAULT nextval('seq_album_id'),
    album_name    VARCHAR(200) NOT NULL,
    genre         VARCHAR(50)  NOT NULL,
    artist_id     INTEGER      NOT NULL REFERENCES Artist(artist_id) ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT chk_album_name  CHECK (LENGTH(TRIM(album_name)) > 0),
    CONSTRAINT chk_album_genre CHECK (LENGTH(TRIM(genre)) > 0)
);
ALTER SEQUENCE seq_album_id OWNED BY Album.album_id;


CREATE TABLE Composition (
    composition_id   INTEGER      PRIMARY KEY DEFAULT nextval('seq_composition_id'),
    composition_name VARCHAR(200) NOT NULL,
    duration         INTEGER      NOT NULL,
    album_id         INTEGER      NOT NULL REFERENCES Album(album_id) ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT chk_composition_name CHECK (LENGTH(TRIM(composition_name)) > 0),
    CONSTRAINT chk_composition_dur  CHECK (duration > 0)
);
ALTER SEQUENCE seq_composition_id OWNED BY Composition.composition_id;
